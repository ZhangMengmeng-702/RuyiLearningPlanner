### 题1：学生成绩管理系统（难度：⭐⭐）

**题目**：实现一个简单的学生成绩管理系统，要求：

功能：
1. 添加学生（姓名、学号、各科成绩）
2. 删除学生（按学号）
3. 修改学生成绩
4. 查询学生信息（按姓名或学号）
5. 显示所有学生成绩，按总分排序
6. 计算班级平均分、最高分、最低分
7. 数据保存到文件（JSON格式），下次启动自动加载

要求：
- 使用面向对象编程（Student 类 + GradeManager 类）
- 使用 JSON 文件持久化存储
- 提供命令行菜单交互

**提示**：
- `json.dump()` 保存，`json.load()` 读取
- 学生用字典或类存储，整个系统用一个列表

**参考解答框架**：
```python
import json
import os

class Student:
    def __init__(self, name, stu_id, scores):
        self.name = name
        self.stu_id = stu_id
        self.scores = scores  # 字典：{"语文": 85, "数学": 92, ...}
    
    def total(self):
        return sum(self.scores.values())
    
    def average(self):
        return self.total() / len(self.scores)
    
    def to_dict(self):
        return {
            "name": self.name,
            "stu_id": self.stu_id,
            "scores": self.scores
        }
    
    @classmethod
    def from_dict(cls, d):
        return cls(d["name"], d["stu_id"], d["scores"])

class GradeManager:
    def __init__(self, filename="grades.json"):
        self.filename = filename
        self.students = []
        self.load()
    
    def add_student(self, student):
        # 检查学号是否重复
        for s in self.students:
            if s.stu_id == student.stu_id:
                return False
        self.students.append(student)
        self.save()
        return True
    
    def remove_student(self, stu_id):
        for i, s in enumerate(self.students):
            if s.stu_id == stu_id:
                del self.students[i]
                self.save()
                return True
        return False
    
    def find_student(self, keyword):
        results = []
        for s in self.students:
            if keyword in s.name or keyword == s.stu_id:
                results.append(s)
        return results
    
    def sorted_by_total(self):
        return sorted(self.students, key=lambda s: s.total(), reverse=True)
    
    def class_stats(self):
        if not self.students:
            return None
        totals = [s.total() for s in self.students]
        return {
            "count": len(self.students),
            "avg_total": sum(totals) / len(totals),
            "max_total": max(totals),
            "min_total": min(totals)
        }
    
    def save(self):
        data = [s.to_dict() for s in self.students]
        with open(self.filename, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
    
    def load(self):
        if os.path.exists(self.filename):
            with open(self.filename, "r", encoding="utf-8") as f:
                data = json.load(f)
            self.students = [Student.from_dict(d) for d in data]

# 主程序（菜单交互）
def main():
    manager = GradeManager()
    
    while True:
        print("\n===== 学生成绩管理系统 =====")
        print("1. 添加学生")
        print("2. 删除学生")
        print("3. 修改成绩")
        print("4. 查询学生")
        print("5. 显示所有学生（按总分排序）")
        print("6. 班级统计")
        print("0. 退出")
        
        choice = input("请选择：")
        
        if choice == "1":
            name = input("姓名：")
            stu_id = input("学号：")
            chinese = int(input("语文："))
            math = int(input("数学："))
            english = int(input("英语："))
            student = Student(name, stu_id, {
                "语文": chinese, "数学": math, "英语": english
            })
            if manager.add_student(student):
                print("添加成功！")
            else:
                print("学号已存在")
        
        elif choice == "2":
            stu_id = input("要删除的学号：")
            if manager.remove_student(stu_id):
                print("删除成功")
            else:
                print("未找到该学生")
        
        elif choice == "3":
            stu_id = input("要修改的学号：")
            results = manager.find_student(stu_id)
            if results:
                s = results[0]
                print(f"当前成绩：{s.scores}")
                subject = input("要修改的科目：")
                new_score = int(input("新分数："))
                s.scores[subject] = new_score
                manager.save()
                print("修改成功")
            else:
                print("未找到该学生")
        
        elif choice == "4":
            keyword = input("输入姓名或学号：")
            results = manager.find_student(keyword)
            if results:
                for s in results:
                    print(f"{s.stu_id} {s.name} 总分:{s.total()} 平均分:{s.average():.1f}")
                    print(f"  各科：{s.scores}")
            else:
                print("未找到")
        
        elif choice == "5":
            sorted_students = manager.sorted_by_total()
            for i, s in enumerate(sorted_students, 1):
                print(f"{i}. {s.stu_id} {s.name} 总分:{s.total()}")
        
        elif choice == "6":
            stats = manager.class_stats()
            if stats:
                print(f"学生人数：{stats['count']}")
                print(f"平均总分：{stats['avg_total']:.1f}")
                print(f"最高总分：{stats['max_total']}")
                print(f"最低总分：{stats['min_total']}")
            else:
                print("暂无学生数据")
        
        elif choice == "0":
            print("再见！")
            break
        
        else:
            print("无效选择")

if __name__ == "__main__":
    main()
```

### 题2：销售数据分析报告（难度：⭐⭐⭐）

**题目**：给定一份销售数据 CSV（自己构造至少 20 行数据），完成一份完整的数据分析报告：

数据字段：日期、产品、地区、销量、单价、销售额

分析要求：
1. 数据概览（行数、列数、数据类型、缺失值检查）
2. 总体销售情况（总销售额、总销量、平均单价）
3. 按产品分析（各产品的总销量、总销售额、占比）
4. 按地区分析（各地区的销售对比）
5. 按月度趋势分析（销售额变化趋势）
6. Top 10 最高销售额的记录
7. 生成可视化图表（柱状图+折线图+饼图，子图布局）
8. 将分析结果保存为文本报告

**提示**：
- 用 Pandas 做数据处理
- 用 Matplotlib 做可视化
- 日期列用 `pd.to_datetime()` 转换
- 月份提取用 `.dt.month`

**参考解答框架**：
```python
import pandas as pd
import matplotlib.pyplot as plt
from datetime import datetime

plt.rcParams["font.sans-serif"] = ["SimHei", "Microsoft YaHei"]
plt.rcParams["axes.unicode_minus"] = False

# 1. 生成模拟数据（或读取真实CSV）
def generate_sample_data():
    import numpy as np
    np.random.seed(42)
    
    dates = pd.date_range("2024-01-01", "2024-06-30", periods=50)
    products = ["产品A", "产品B", "产品C", "产品D"]
    regions = ["华东", "华北", "华南", "西南", "东北"]
    
    data = []
    for d in dates:
        product = np.random.choice(products)
        region = np.random.choice(regions)
        quantity = np.random.randint(10, 200)
        price = np.random.choice([50, 80, 120, 200])
        amount = quantity * price
        data.append({
            "日期": d.strftime("%Y-%m-%d"),
            "产品": product,
            "地区": region,
            "销量": quantity,
            "单价": price,
            "销售额": amount
        })
    
    df = pd.DataFrame(data)
    df.to_csv("sales_data.csv", index=False, encoding="utf-8-sig")
    return df

# 2. 读取数据
df = generate_sample_data()  # 或 df = pd.read_csv("sales_data.csv")
df["日期"] = pd.to_datetime(df["日期"])
df["月份"] = df["日期"].dt.month

# 3. 数据分析
report = []
report.append("=" * 50)
report.append("销售数据分析报告")
report.append(f"生成时间：{datetime.now().strftime('%Y-%m-%d %H:%M:%S')}")
report.append("=" * 50)

# 数据概览
report.append("\n【1. 数据概览】")
report.append(f"数据行数：{len(df)}")
report.append(f"数据列数：{len(df.columns)}")
report.append(f"列名：{df.columns.tolist()}")
report.append(f"缺失值数量：\n{df.isnull().sum()}")

# 总体情况
report.append("\n【2. 总体销售情况】")
report.append(f"总销售额：{df['销售额'].sum():,.2f} 元")
report.append(f"总销量：{df['销量'].sum()} 件")
report.append(f"平均单价：{df['单价'].mean():.2f} 元")
report.append(f"交易笔数：{len(df)} 笔")
report.append(f"单笔平均销售额：{df['销售额'].mean():.2f} 元")

# 按产品分析
report.append("\n【3. 按产品分析】")
product_stats = df.groupby("产品").agg({
    "销量": "sum",
    "销售额": "sum"
}).sort_values("销售额", ascending=False)
product_stats["销售额占比"] = product_stats["销售额"] / product_stats["销售额"].sum() * 100
report.append(product_stats.to_string())

# 按地区分析
report.append("\n【4. 按地区分析】")
region_stats = df.groupby("地区")["销售额"].sum().sort_values(ascending=False)
report.append(region_stats.to_string())

# 月度趋势
report.append("\n【5. 月度销售趋势】")
monthly = df.groupby("月份")["销售额"].sum()
report.append(monthly.to_string())

# Top 10
report.append("\n【6. 销售额 Top 10】")
top10 = df.nlargest(10, "销售额")[["日期", "产品", "地区", "销售额"]]
report.append(top10.to_string(index=False))

# 保存报告
report_text = "\n".join(report)
with open("sales_report.txt", "w", encoding="utf-8") as f:
    f.write(report_text)
print("报告已保存到 sales_report.txt")

# 4. 可视化
fig, axes = plt.subplots(2, 2, figsize=(14, 10))
fig.suptitle("销售数据分析", fontsize=20)

# 产品销售柱状图
ax1 = axes[0, 0]
product_stats["销售额"].plot(kind="bar", ax=ax1, color="#4ECDC4")
ax1.set_title("各产品销售额")
ax1.set_ylabel("销售额（元）")
ax1.tick_params(axis="x", rotation=0)

# 地区饼图
ax2 = axes[0, 1]
ax2.pie(region_stats.values, labels=region_stats.index,
        autopct="%1.1f%%", startangle=90)
ax2.set_title("各地区销售额占比")
ax2.axis("equal")

# 月度趋势折线图
ax3 = axes[1, 0]
monthly.plot(kind="line", ax=ax3, marker="o", color="#FF6B6B", linewidth=2)
ax3.set_title("月度销售趋势")
ax3.set_xlabel("月份")
ax3.set_ylabel("销售额（元）")
ax3.grid(True, alpha=0.3)

# 销量与销售额散点图
ax4 = axes[1, 1]
scatter = ax4.scatter(df["销量"], df["销售额"], c=df["单价"],
                      cmap="viridis", alpha=0.7)
plt.colorbar(scatter, ax=ax4, label="单价")
ax4.set_title("销量 vs 销售额")
ax4.set_xlabel("销量（件）")
ax4.set_ylabel("销售额（元）")
ax4.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig("sales_analysis.png", dpi=100)
print("图表已保存到 sales_analysis.png")
plt.show()
```

### 题3：简易记事本（难度：⭐⭐）

**题目**：用 Python 写一个带 GUI 的简易记事本（可选，学有余力做），或者命令行版本：

功能要求：
1. 新建、打开、保存、另存为文本文件
2. 支持查找、替换文字
3. 显示行数、字数统计
4. 自动保存（可选）
5. 最近打开文件列表（可选）

如果做 GUI 版，使用 `tkinter`（Python 内置库）。

**参考框架（tkinter 版）**：
```python
import tkinter as tk
from tkinter import filedialog, messagebox, simpledialog
import os

class Notepad:
    def __init__(self, root):
        self.root = root
        self.root.title("简易记事本")
        self.root.geometry("800x600")
        
        self.current_file = None
        self.recent_files = []
        
        self._create_menu()
        self._create_text_area()
        self._create_status_bar()
    
    def _create_menu(self):
        menubar = tk.Menu(self.root)
        
        # 文件菜单
        file_menu = tk.Menu(menubar, tearoff=0)
        file_menu.add_command(label="新建", command=self.new_file, accelerator="Ctrl+N")
        file_menu.add_command(label="打开", command=self.open_file, accelerator="Ctrl+O")
        file_menu.add_command(label="保存", command=self.save_file, accelerator="Ctrl+S")
        file_menu.add_command(label="另存为", command=self.save_as)
        file_menu.add_separator()
        file_menu.add_command(label="退出", command=self.root.quit)
        menubar.add_cascade(label="文件", menu=file_menu)
        
        # 编辑菜单
        edit_menu = tk.Menu(menubar, tearoff=0)
        edit_menu.add_command(label="查找", command=self.find_text, accelerator="Ctrl+F")
        edit_menu.add_command(label="替换", command=self.replace_text, accelerator="Ctrl+H")
        edit_menu.add_separator()
        edit_menu.add_command(label="全选", command=self.select_all, accelerator="Ctrl+A")
        menubar.add_cascade(label="编辑", menu=edit_menu)
        
        # 帮助菜单
        help_menu = tk.Menu(menubar, tearoff=0)
        help_menu.add_command(label="关于", command=self.show_about)
        menubar.add_cascade(label="帮助", menu=help_menu)
        
        self.root.config(menu=menubar)
        
        # 绑定快捷键
        self.root.bind("<Control-n>", lambda e: self.new_file())
        self.root.bind("<Control-o>", lambda e: self.open_file())
        self.root.bind("<Control-s>", lambda e: self.save_file())
        self.root.bind("<Control-f>", lambda e: self.find_text())
        self.root.bind("<Control-h>", lambda e: self.replace_text())
        self.root.bind("<Control-a>", lambda e: self.select_all())
    
    def _create_text_area(self):
        self.text = tk.Text(self.root, wrap="word", font=("微软雅黑", 12))
        self.text.pack(fill="both", expand=True)
        self.text.bind("<KeyRelease>", self.update_status)
        
        scrollbar = tk.Scrollbar(self.text)
        scrollbar.pack(side="right", fill="y")
        self.text.config(yscrollcommand=scrollbar.set)
        scrollbar.config(command=self.text.yview)
    
    def _create_status_bar(self):
        self.status = tk.Label(self.root, text="就绪", bd=1, relief="sunken", anchor="w")
        self.status.pack(side="bottom", fill="x")
    
    def new_file(self):
        self.text.delete("1.0", "end")
        self.current_file = None
        self.root.title("无标题 - 简易记事本")
        self.update_status()
    
    def open_file(self):
        filename = filedialog.askopenfilename(
            filetypes=[("文本文件", "*.txt"), ("所有文件", "*.*")]
        )
        if filename:
            try:
                with open(filename, "r", encoding="utf-8") as f:
                    content = f.read()
                self.text.delete("1.0", "end")
                self.text.insert("1.0", content)
                self.current_file = filename
                self.root.title(f"{os.path.basename(filename)} - 简易记事本")
                self.update_status()
            except Exception as e:
                messagebox.showerror("错误", f"打开文件失败：{str(e)}")
    
    def save_file(self):
        if self.current_file:
            try:
                content = self.text.get("1.0", "end-1c")
                with open(self.current_file, "w", encoding="utf-8") as f:
                    f.write(content)
                self.status.config(text=f"已保存：{self.current_file}")
                return True
            except Exception as e:
                messagebox.showerror("错误", f"保存失败：{str(e)}")
                return False
        else:
            return self.save_as()
    
    def save_as(self):
        filename = filedialog.asksaveasfilename(
            defaultextension=".txt",
            filetypes=[("文本文件", "*.txt"), ("所有文件", "*.*")]
        )
        if filename:
            self.current_file = filename
            self.root.title(f"{os.path.basename(filename)} - 简易记事本")
            return self.save_file()
        return False
    
    def find_text(self):
        find_str = simpledialog.askstring("查找", "请输入要查找的内容：")
        if find_str:
            self.text.tag_remove("found", "1.0", "end")
            idx = "1.0"
            count = 0
            while True:
                idx = self.text.search(find_str, idx, nocase=True, stopindex="end")
                if not idx:
                    break
                end_idx = f"{idx}+{len(find_str)}c"
                self.text.tag_add("found", idx, end_idx)
                idx = end_idx
                count += 1
            self.text.tag_config("found", background="yellow")
            self.status.config(text=f"找到 {count} 处")
    
    def replace_text(self):
        find_str = simpledialog.askstring("替换", "查找内容：")
        if not find_str:
            return
        replace_str = simpledialog.askstring("替换", "替换为：")
        if replace_str is None:
            return
        
        content = self.text.get("1.0", "end-1c")
        new_content = content.replace(find_str, replace_str)
        count = content.count(find_str)
        
        if count > 0:
            self.text.delete("1.0", "end")
            self.text.insert("1.0", new_content)
            self.status.config(text=f"已替换 {count} 处")
        else:
            self.status.config(text="未找到匹配内容")
    
    def select_all(self):
        self.text.tag_add("sel", "1.0", "end")
        return "break"
    
    def update_status(self, event=None):
        content = self.text.get("1.0", "end-1c")
        lines = len(content.split("\n"))
        chars = len(content)
        words = len(content.split())
        self.status.config(text=f"行数：{lines} | 字符数：{chars} | 单词数：{words}")
    
    def show_about(self):
        messagebox.showinfo("关于", "简易记事本 v1.0\n\n一个简单的 Python tkinter 记事本")
    
    def on_close(self):
        if messagebox.askyesno("退出", "确定要退出吗？"):
            self.root.destroy()

if __name__ == "__main__":
    root = tk.Tk()
    app = Notepad(root)
    root.protocol("WM_DELETE_WINDOW", app.on_close)
    root.mainloop()
```
