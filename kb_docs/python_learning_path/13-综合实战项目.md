## 项目背景

你拿到了某电商平台 2024 年 1 月的销售数据（CSV 格式），需要分析：

1. 整体销售情况（总销售额、总订单数、客单价）
2. 各品类销售排名
3. 每日销售额趋势
4. 用户复购率分析
5. 输出一份包含图表和结论的分析报告

# sales_2024_01.csv 的数据字段

## 完整代码（项目骨架）

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from datetime import datetime


class SalesAnalyzer:
    """电商销售数据分析器"""

    def __init__(self, filepath: str):
        self.df = pd.read_csv(filepath, encoding="utf-8")
        self._clean_data()
        self._enrich_data()

    def _clean_data(self):
        """数据清洗"""
        # 删除缺失值
        self.df.dropna(inplace=True)
        # 删除重复订单
        self.df.drop_duplicates(subset=["order_id"], inplace=True)
        # 金额必须为正
        self.df = self.df[self.df["amount"] > 0]

    def _enrich_data(self):
        """衍生字段"""
        self.df["order_date"] = pd.to_datetime(self.df["order_date"])
        self.df["month"] = self.df["order_date"].dt.month
        self.df["day"] = self.df["order_date"].dt.day
        self.df["weekday"] = self.df["order_date"].dt.day_name()

    def overview(self) -> dict:
        """整体概览"""
        return {
            "total_orders": len(self.df),
            "total_sales": round(self.df["amount"].sum(), 2),
            "avg_order_value": round(self.df["amount"].mean(), 2),
            "total_users": self.df["user_id"].nunique(),
            "categories": self.df["category"].nunique(),
        }

    def sales_by_category(self):
        """各品类销售额"""
        return self.df.groupby("category")["amount"].agg(["sum", "mean", "count"])\
            .sort_values("sum", ascending=False)

    def daily_sales_trend(self):
        """每日销售额趋势"""
        return self.df.groupby("order_date")["amount"].sum()

    def repurchase_rate(self) -> float:
        """用户复购率（购买 ≥2 次的用户占比）"""
        user_counts = self.df["user_id"].value_counts()
        repeat = (user_counts >= 2).sum()
        total = len(user_counts)
        return round(repeat / total * 100, 2)

    def plot_overview(self):
        """生成可视化报告"""
        fig, axes = plt.subplots(2, 2, figsize=(14, 10))

        # 1. 各品类销售额柱状图
        cat_data = self.sales_by_category()
        axes[0, 0].bar(cat_data.index, cat_data["sum"])
        axes[0, 0].set_title("各品类销售额")
        axes[0, 0].tick_params(axis="x", rotation=45)

        # 2. 每日销售额趋势
        daily = self.daily_sales_trend()
        axes[0, 1].plot(daily.index, daily.values, marker="o", markersize=3)
        axes[0, 1].set_title("每日销售额趋势")
        axes[0, 1].tick_params(axis="x", rotation=45)

        # 3. 客单价分布直方图
        axes[1, 0].hist(self.df["amount"], bins=30, edgecolor="white")
        axes[1, 0].set_title("客单价分布")
        axes[1, 0].set_xlabel("金额（元）")

        # 4. 各品类订单量占比（饼图）
        order_counts = self.df["category"].value_counts()
        axes[1, 1].pie(order_counts.values, labels=order_counts.index,
                       autopct="%1.1f%%")
        axes[1, 1].set_title("各品类订单量占比")

        plt.tight_layout()
        plt.savefig("sales_report.png", dpi=200, bbox_inches="tight")
        plt.show()

    def generate_report(self) -> str:
        """生成文字分析报告"""
        ov = self.overview()
        top_cat = self.sales_by_category().head(3)

        report = f"""
# 电商销售数据分析报告（2024年1月）

## 整体概况
- 总订单数：{ov['total_orders']}
- 总销售额：{ov['total_sales']} 元
- 平均客单价：{ov['avg_order_value']} 元
- 活跃用户数：{ov['total_users']}
- 商品品类数：{ov['categories']}

## 销售排名
{top_cat.to_string()}

## 复购率
{self.repurchase_rate()}% 的用户产生了至少 2 次购买。

## 结论与建议
（基于以上数据，给出具体的业务建议）
"""
        return report


# 执行入口
if __name__ == "__main__":
    analyzer = SalesAnalyzer("sales_2024_01.csv")
    print(analyzer.generate_report())
    analyzer.plot_overview()
```

## 常见问题

**Q**: 没有真实数据怎么办？
**A**: 用 Python 生成模拟数据：
```python
import random, pandas as pd
categories = ["电子产品", "服装", "食品", "家居", "图书"]
data = []
for i in range(1000):
    data.append({
        "order_id": f"ORD{i+1:05d}",
        "user_id": f"USR{random.randint(1, 200):03d}",
        "category": random.choice(categories),
        "amount": round(random.uniform(10, 2000), 2),
        "quantity": random.randint(1, 5),
        "order_date": f"2024-01-{random.randint(1, 31):02d}",
    })
pd.DataFrame(data).to_csv("sales_2024_01.csv", index=False, encoding="utf-8")
```
